export interface FetchBooksOptions {
  /** true => giả lập API thất bại */
  shouldFail: boolean;
  /** Độ trễ giả lập (ms) */
  delay: number;
}
 
const DEFAULT_OPTIONS: FetchBooksOptions = { shouldFail: false, delay: 500 };
 
/**
 * API giả lập lấy danh sách sách.
 * Luôn resolve về ApiResult<Book[]> (không throw) để phía gọi dùng type guard phân biệt.
 */
export function fetchBooksApi(options: Partial<FetchBooksOptions> = {}): Promise<ApiResult<Book[]>> {
  const { shouldFail, delay } = { ...DEFAULT_OPTIONS, ...options };
 
  return new Promise((resolve) => {
    setTimeout(() => {
      if (shouldFail) {
        resolve({ ok: false, error: 'Không thể tải danh sách sách. Vui lòng thử lại!' });
        return;
      }
      // Trả bản sao để tránh bị sửa dữ liệu gốc
      resolve({ ok: true, data: MOCK_BOOKS.map((b) => ({ ...b })) });
    }, delay);
  });
}
